cmake_minimum_required(VERSION 3.16)
project(heron_recorder LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(UHD REQUIRED HINTS /opt/uhd/lib/cmake/uhd /usr/lib/x86_64-linux-gnu/cmake/uhd /usr/lib/aarch64-linux-gnu/cmake/uhd)
find_package(Boost REQUIRED)
find_package(Threads REQUIRED)
add_executable(heron-recorder main.cpp)
target_include_directories(heron-recorder PRIVATE ${UHD_INCLUDE_DIRS} ${Boost_INCLUDE_DIRS})
target_link_libraries(heron-recorder PRIVATE ${UHD_LIBRARIES} Threads::Threads)
target_compile_options(heron-recorder PRIVATE -Wall -Wextra -O2)
set_target_properties(heron-recorder PROPERTIES INSTALL_RPATH "/opt/uhd/lib")
install(TARGETS heron-recorder RUNTIME DESTINATION bin)
